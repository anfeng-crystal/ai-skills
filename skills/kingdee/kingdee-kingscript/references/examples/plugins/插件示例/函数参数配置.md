## 使用范围与消费合同

两个案例均是公式参数动态表单：继承 `AbstractFormPlugin`、实现 `IFuncParamInputFormPlugin`，由实际表单模型提供字段值；后者继承 `IFuncParamEditor`。接口没有 `getParamValue`，也没有通用 `setSetting` 回填方法。字段标识沿用本例，需在实际参数页面元数据中配置，不能直接把 F7 对象或数字字段当成文本。

`checkSetting(msgBuilder): boolean` 把错误写入 `StringBuilder`；`getSetting(): string` 返回要插入公式编辑器的完整函数表达式。实际参数容器可能在校验失败后继续调用 `getSetting`，所以两个方法共用校验，非法时后者返回空串。接口 getter 和设计/运行注册的完整说明见[公式平台插件](./公式平台插件.md)。

标准函数树入口把函数定义 `FuncInfo` 传给参数容器；参数子页从当前模型读用户输入，容器确认后将返回字符串交给设计器的 `insertExpression`。这条链路不等于保存业务数据，也未提供既有某次调用 JSON 的通用恢复入口。已有设置回显应在已核实的项目入口解析并回填模型，不能用新建默认值覆盖历史配置。

本例分别约定 `AGING_ANALYSIS(configJson)`、`STOCK_ALERT(configJson)` 接收一个 JSON **字符串**参数；这些名称和 JSON 结构属于示例协议，需要对应自定义函数提供者接入、分别完成设计期/运行期注册。参数页只负责配置与表达式生成，实际查询、计算、权限、保存和通知由项目消费者实现。JSON 双层转义在本地可验证，目标公式解析器及 KingScript 接口桥接仍须平台验收。

## checkSetting - 校验函数参数配置

### 业务场景：账龄分析区间

保留实体、日期字段、金额汇总字段与账龄区间配置。本例按**整数天、覆盖全部非负账龄**约定：首段从 `0` 开始，有限区间为闭区间且 `end > start`，下一段 `start = 上段 end + 1`，仅最后一段允许 `end=-1` 表示无上限。例如 `0–30、31–60、61–90、91天及以上`；单日有限段不在本例原有规则内。首段从 0 是本例覆盖口径，不是平台接口强制规则。

模型字段 `entityName`、`dateField`、`amountField` 保存真实实体/字段标识的文本，`agingIntervals` 保存 JSON 数组文本。先检查原始 JSON 中所有数字的写法，再检查形状与区间逻辑。本例只接受安全范围内的十进制整数字面量，拒绝小数和指数写法（包括 `0.0`、`0e0`），`-0` 按零接受；字符串内容不受数字词法检查影响。这比数值是整数更严格，目的是避免 JSON 解析时近整数舍入或小数下溢后混入区间。它属于本例输入协议，不是平台通用 JSON 限制；不依靠 TypeScript `as` 在运行时转换或校验数据。

```typescript
import { AbstractFormPlugin } from "@cosmic/bos-core/kd/bos/form/plugin";
import { IFuncParamInputFormPlugin } from "@cosmic/bos-core/kd/bos/formula/platform/api/funcpara";

/** 项目参数协议：账龄按整数天划分，闭区间从 0 连续覆盖，最后一段 end=-1 表示无上限。 */
export interface AgingRange { start: number; end: number; }
export interface AgingInput {
  entityName: string;
  dateField: string;
  amountField: string;
  agingIntervals: string;
}
export interface AgingSetting {
  entityName: string;
  dateField: string;
  amountField: string;
  agingIntervals: AgingRange[];
}
export interface AgingValidation { setting: AgingSetting | null; errors: string[]; }

/** JSON 数字须为安全整数，避免小数天、隐式字符串转换或超出 JS 精度的端点进入连续性判断。 */
function isSafeInteger(value: unknown): value is number {
  return typeof value === "number" && isFinite(value) &&
    Math.floor(value) === value && Math.abs(value) <= 9007199254740991;
}

/**
 * 本例区间 JSON 的所有数字只接受十进制整数字面量，不接受小数或指数写法（即使其数值等于整数）。
 * 在 JSON.parse 前逐 token 检查，防止近整数舍入或极小数下溢后伪装成整数；字符串及转义字符原样跳过。
 * 不根据 start/end 的原文拼写找数字，因此 Unicode 转义键、重复键也不能绕过数字规则。
 * 这里只检查数字词法与安全范围，完整 JSON 语法仍由 JSON.parse 检查。
 */
function jsonIntegerTokenError(text: string): string | null {
  let i = 0;
  while (i < text.length) {
    const ch = text.charAt(i);
    if (ch === '"') {
      i++;
      while (i < text.length) {
        const current = text.charAt(i);
        if (current === "\\") { i += 2; continue; }
        i++;
        if (current === '"') break;
      }
      continue;
    }
    if (ch === "-" || (ch >= "0" && ch <= "9")) {
      const begin = i++;
      while (i < text.length && /[0-9eE+.\-]/.test(text.charAt(i))) i++;
      const token = text.slice(begin, i);
      if (!/^-?(0|[1-9][0-9]*)$/.test(token)) {
        return "区间 JSON 的数字必须使用十进制整数字面量，不允许小数或指数写法";
      }
      const digits = token.charAt(0) === "-" ? token.slice(1) : token;
      const maximum = "9007199254740991";
      if (digits.length > maximum.length || (digits.length === maximum.length && digits > maximum)) {
        return "区间 JSON 的数字不能超过安全整数范围";
      }
      continue;
    }
    i++;
  }
  return null;
}

/**
 * 同时校验配置项和区间，失败返回全部可安全判断的错误，不抛 JSON 解析异常。
 * entityName/dateField/amountField 的存在、实际字段类型和访问权限由项目接入层校验；本函数仅检查非空。
 */
export function validateAgingInput(input: AgingInput): AgingValidation {
  const errors: string[] = [];
  const entityName = input.entityName.trim();
  const dateField = input.dateField.trim();
  const amountField = input.amountField.trim();
  if (entityName === "") errors.push("请选择应收/应付单据类型");
  if (dateField === "") errors.push("请选择日期字段");
  if (amountField === "") errors.push("请选择金额汇总字段");

  let parsed: unknown = null;
  let parsedSuccessfully = false;
  if (input.agingIntervals.trim() === "") {
    errors.push("请至少配置一个账龄区间");
  } else {
    const tokenError = jsonIntegerTokenError(input.agingIntervals);
    if (tokenError !== null) errors.push(tokenError);
    else {
      try { parsed = JSON.parse(input.agingIntervals); parsedSuccessfully = true; }
      catch (_error) { errors.push("账龄区间必须是合法的 JSON 数组"); }
    }
    if (parsedSuccessfully && (!Array.isArray(parsed) || parsed.length === 0)) {
      errors.push("账龄区间必须是至少包含一个区间的 JSON 数组");
    }
  }

  const ranges: AgingRange[] = [];
  if (Array.isArray(parsed)) {
    for (let i = 0; i < parsed.length; i++) {
      const item: unknown = parsed[i];
      const prefix = "第" + (i + 1) + "个区间";
      if (item === null || typeof item !== "object" || Array.isArray(item)) {
        errors.push(prefix + "必须是包含 start 和 end 的对象");
        continue;
      }
      const current = item as { start?: unknown; end?: unknown };
      const start = current.start;
      const end = current.end;
      if (!isSafeInteger(start) || start < 0) {
        errors.push(prefix + "的起始天数必须是非负安全整数");
      }
      if (!isSafeInteger(end) || end < -1) {
        errors.push(prefix + "的结束天数必须是非负安全整数或 -1");
      }
      if (!isSafeInteger(start) || !isSafeInteger(end)) continue;

      if (i === 0 && start !== 0) errors.push("首个账龄区间必须从 0 天开始");
      if (end === -1 && i !== parsed.length - 1) errors.push(prefix + "不是末尾区间，不能设置无上限");
      if (end !== -1 && end <= start) errors.push(prefix + "的结束天数必须大于起始天数");
      if (i > 0) {
        const previous: unknown = parsed[i - 1];
        if (previous !== null && typeof previous === "object" && !Array.isArray(previous)) {
          const prevEnd = (previous as { end?: unknown }).end;
          if (isSafeInteger(prevEnd) && prevEnd !== -1 && start !== prevEnd + 1) {
            errors.push("第" + i + "和第" + (i + 1) + "个区间不连续，存在间断或重叠");
          }
        }
      }
      if (i === parsed.length - 1 && end !== -1) errors.push("最后一个区间必须为开放区间（end=-1）");
      ranges.push({ start, end });
    }
  }
  return {
    setting: errors.length === 0 ? { entityName, dateField, amountField, agingIntervals: ranges } : null,
    errors
  };
}

/**
 * 返回本例约定的函数表达式，唯一参数为 JSON 字符串，先序列化配置再转义为字符串字面量。
 * AGING_ANALYSIS 为待接入的自定义函数；provider 须声明一项字符串参数，并自行解析、校验和计算。
 * 本方法不注册函数、不执行公式，也不实现数据查询和持久化。
 */
export function buildAgingExpression(setting: AgingSetting): string {
  return "AGING_ANALYSIS(" + JSON.stringify(JSON.stringify(setting)) + ")";
}

/**
 * 账龄参数页插件；动态表单须配置文本字段 entityName/dateField/amountField/agingIntervals。
 * 前三个字段保存实体及字段标识，agingIntervals 保存区间 JSON 文本；字段名须与实际元数据一致。
 * 参数来源仅为当前表单 model。已有配置怎样传入及回填取决于项目入口，不能从接口上下文 getter 猜测。
 * getSetting 返回 AGING_ANALYSIS("JSON字符串")；JSON 结构是项目协议，不能裸传 JSON 冒充公式表达式。
 * 项目必须注册对应自定义函数；本例不表示已自动持久化或执行账龄分析。
 */
class AgingAnalysisParameterPlugin extends AbstractFormPlugin implements IFuncParamInputFormPlugin {
  /** 当前表单值经相同规则校验，供容器展示错误信息。 */
  checkSetting(msgBuilder: $.java.lang.StringBuilder): boolean {
    const result = validateAgingInput(this.readInput());
    for (const error of result.errors) msgBuilder.append(error + "；");
    return result.setting !== null;
  }

  /** 容器仍可能继续调用本方法；每次重新校验，非法输入返回空串，合法输入才返回函数表达式。 */
  getSetting(): string {
    const result = validateAgingInput(this.readInput());
    return result.setting === null ? "" : buildAgingExpression(result.setting);
  }

  /** 示例以文本字段保存标识和 JSON；不把基础资料对象假定为字符串 ID。 */
  private readInput(): AgingInput {
    const model = this.getModel();
    const text = (key: string): string => {
      const value = model.getValue(key);
      return value == null ? "" : String(value);
    };
    return {
      entityName: text("entityName"), dateField: text("dateField"),
      amountField: text("amountField"), agingIntervals: text("agingIntervals")
    };
  }
}
const plugin = new AgingAnalysisParameterPlugin();
export { plugin };
```

**配置输入示例**：

```json
[{"start":0,"end":30},{"start":31,"end":60},{"start":61,"end":90},{"start":91,"end":-1}]
```

输出 payload 同时包含 `entityName/dateField/amountField/agingIntervals`，再作为一个字符串参数包装进 `AGING_ANALYSIS(...)`。不把校验通过理解为实体、字段类型或访问权限已经验证；消费方必须按实际元数据核对日期/金额字段，明确账龄基准日和统计范围。

## getSetting - 获取函数参数配置

### 业务场景：库存预警参数

保留仓库 ID/名称，`all/category/specific` 三种物料范围，可选最低/最高库存，`message/email/both` 三种通知方式及收件人。所有输入字段按示例约定为文本；主键与十进制值始终保留字符串，不经过 JS `Number`。F7、Decimal 或分录名单控件应按其实际返回类型适配。

空枚举继续默认 `all/message`；只填写一侧阈值也要验证格式，两侧都填时用 `BigDecimal.compareTo` 检查上下界。未填阈值省略，`"0"` 保留；沿用有符号值、科学计数法和任意精度文本，不擅加非负、舍入或截断规则。是否允许负库存阈值由实际业务合同决定。

分类范围需要单个分类 ID，指定物料范围需要逗号分隔且无空项的 ID 名单。ID 为不透明文本，不猜数字主键。本例明确要求三种通知方式都填写收件人名单；若实际消费者支持空名单，应按其明确语义调整，不能隐含补为当前用户。这里只验证配置形状，不证明对象存在、邮箱可用或用户有权限。

```typescript
import { AbstractFormPlugin } from "@cosmic/bos-core/kd/bos/form/plugin";
import { IFuncParamInputFormPlugin } from "@cosmic/bos-core/kd/bos/formula/platform/api/funcpara";
import { BigDecimal } from "@cosmic/bos-script/java/math";

/** 面板字段均为文本值；主键和十进制值不经过 JS Number。 */
interface StockAlertConfig {
  warehouseId: string;
  warehouseName: string;
  materialScope: string;
  materialCategoryId: string;
  materialIds: string;
  minQty: string;
  maxQty: string;
  notifyType: string;
  notifyUserIds: string;
}

/** ID 是不透明字符串，逗号仅用作名单分隔符；不推断主键的数字类型。 */
function idError(value: string): boolean {
  return value.trim() === "" || value.indexOf(",") >= 0;
}

/** 名单至少有一个 ID；拒绝空项，保留长整数、字母和 UUID 的字符串表示。 */
function idListError(value: string): boolean {
  return value.split(",").some(id => idError(id));
}

/** 仅清理名单分隔符旁空白，不转数字、不截断主键。 */
function normalizedIds(value: string): string {
  return value.split(",").map(id => id.trim()).join(",");
}

/**
 * 同时供 checkSetting/getSetting 使用。空枚举采用 all/message，未知枚举拒绝。
 * 每个已填阈值都必须能构造 BigDecimal；至少一侧已填，双侧时 min <= max。
 * 沿用原例的有符号阈值能力；是否禁止负数由实际业务合同决定。
 */
function stockConfigError(value: StockAlertConfig): string | null {
  if (idError(value.warehouseId)) return "请选择目标仓库，并填写单个仓库 ID；";
  const scope = value.materialScope.trim() || "all";
  if (scope !== "all" && scope !== "category" && scope !== "specific") {
    return "物料范围只能是 all、category 或 specific；";
  }
  if (scope === "category" && idError(value.materialCategoryId)) return "分类范围必须填写单个物料分类 ID；";
  if (scope === "specific" && idListError(value.materialIds)) return "指定物料范围必须填写无空项的物料 ID 名单；";
  const minText = value.minQty.trim();
  const maxText = value.maxQty.trim();
  if (minText === "" && maxText === "") return "最低库存和最高库存至少填写一项；";
  let min: BigDecimal | null = null;
  let max: BigDecimal | null = null;
  try {
    if (minText !== "") min = new BigDecimal(minText);
    if (maxText !== "") max = new BigDecimal(maxText);
  } catch (_) {
    return "最低库存和最高库存必须是有效的十进制数；";
  }
  if (min != null && max != null && min.compareTo(max) > 0) return "最低库存不能大于最高库存；";
  const notify = value.notifyType.trim() || "message";
  if (notify !== "message" && notify !== "email" && notify !== "both") {
    return "通知方式只能是 message、email 或 both；";
  }
  if (idListError(value.notifyUserIds)) return "请填写无空项的通知收件人 ID 名单；";
  return null;
}

interface StockAlertSetting {
  warehouse: { id: string; name: string };
  materialScope: string;
  materialCategoryId?: string;
  materialIds?: string;
  threshold: { min?: string; max?: string };
  notify: { type: string; userIds: string };
}

/**
 * 生成本例约定的配置 JSON；非法输入返回空串。只保存当前范围对应的 ID 字段，
 * 不会将 all 范围中残留的分类/物料名单误传给消费者；未填阈值省略，数值 0 保留。
 */
function stockSetting(value: StockAlertConfig): string {
  if (stockConfigError(value) != null) return "";
  const scope = value.materialScope.trim() || "all";
  const setting: StockAlertSetting = {
    warehouse: { id: value.warehouseId.trim(), name: value.warehouseName },
    materialScope: scope,
    threshold: {},
    notify: { type: value.notifyType.trim() || "message", userIds: normalizedIds(value.notifyUserIds) }
  };
  if (scope === "category") setting.materialCategoryId = value.materialCategoryId.trim();
  if (scope === "specific") setting.materialIds = normalizedIds(value.materialIds);
  if (value.minQty.trim() !== "") setting.threshold.min = value.minQty.trim();
  if (value.maxQty.trim() !== "") setting.threshold.max = value.maxQty.trim();
  return JSON.stringify(setting);
}

/**
 * 本例自定义 STOCK_ALERT 函数约定接收一个 JSON 字符串参数。
 * 函数参数编辑器返回完整表达式；JSON.stringify 对 payload 的引号/反斜杠再次转义。
 * 实际函数提供者必须注册 STOCK_ALERT，按上述 JSON 协议解析并再次验证配置。
 */
function stockExpression(value: StockAlertConfig): string {
  const payload = stockSetting(value);
  return payload === "" ? "" : "STOCK_ALERT(" + JSON.stringify(payload) + ")";
}

/**
 * 库存预警的函数参数动态表单插件，字段标识见 StockAlertConfig。
 * warehouseId/materialCategoryId/materialIds/notifyUserIds 为保存 ID 文本的字段，
 * minQty/maxQty 为十进制文本字段；F7、整数或 Decimal 控件须先按其真实返回类型适配。
 * 本例只读取编辑器当前模型并返回函数表达式，不查询库存、不判断权限、不发送通知。
 * 已有配置的回填须由已核实的参数页入口完成，不以新建默认值覆盖历史配置。
 */
class StockAlertFuncParamEditor extends AbstractFormPlugin implements IFuncParamInputFormPlugin {
  /** 错误写入容器提供的 StringBuilder；不依赖容器据此阻断 getSetting。 */
  checkSetting(msgBuilder: $.java.lang.StringBuilder): boolean {
    const error = stockConfigError(this.readConfig());
    if (error == null) return true;
    msgBuilder.append(error);
    return false;
  }

  /** 校验当前模型并返回配置；即使单独调用也不会输出非法设置。 */
  getSetting(): string {
    return stockExpression(this.readConfig());
  }

  /** 空模型文本按空串处理；必填、默认和分支规则统一交由校验/序列化执行。 */
  private readConfig(): StockAlertConfig {
    const model = this.getModel();
    const text = (key: string): string => {
      const value = model.getValue(key);
      return value == null ? "" : String(value);
    };
    return {
      warehouseId: text("warehouseId"), warehouseName: text("warehouseName"),
      materialScope: text("materialScope"), materialCategoryId: text("materialCategoryId"),
      materialIds: text("materialIds"), minQty: text("minQty"), maxQty: text("maxQty"),
      notifyType: text("notifyType"), notifyUserIds: text("notifyUserIds")
    };
  }
}
const plugin = new StockAlertFuncParamEditor();
export { plugin, stockConfigError, stockSetting, stockExpression, StockAlertConfig };
```

**函数参数内的 JSON payload 示例**（不是 `getSetting` 的裸返回值）：

```json
{
  "warehouse": { "id": "100001", "name": "成品仓" },
  "materialScope": "category",
  "materialCategoryId": "200001",
  "threshold": { "min": "100", "max": "5000" },
  "notify": { "type": "both", "userIds": "300001,300002" }
}
```

`stockExpression` 将该 JSON 再转义为字符串字面量，返回 `STOCK_ALERT("转义后的JSON")`。切换范围后仅输出当前分支的分类/物料 ID，避免其他分支残留值被误用；两个可选阈值独立保留。消费者应解析并复验该配置，明确仓库/物料数据范围、库存口径、接收人和执行时机；参数预览或公式编辑期间不应被示例误导为已经执行了通知。
